CREATE PROCEDURE [dbo].[ActualizarContrasena]
(
    @Id BIGINT,
    @Contrasena NVARCHAR(250),
    @ModificadoPor NVARCHAR(25),
    @FechaModificacion DATETIME,
    @Usuario NVARCHAR(25)
)
AS
BEGIN
    -- Validar que el usuario que realiza el cambio pertenezca a la misma empresa
    -- que el usuario cuya contraseña se va a actualizar
    IF NOT EXISTS(
        SELECT 1 
        FROM Usuarios u
        WHERE u.Id = @Id 
            AND u.Estatus = 1
            AND EXISTS(
                SELECT 1 
                FROM Usuarios m
                WHERE m.NombreUsuario = @Usuario 
                    AND m.EmpresaId = u.EmpresaId
                    AND m.Estatus = 1
            )
    )
    BEGIN
        SELECT 0
        RETURN
    END

    UPDATE Usuarios
    SET Contrasena = @Contrasena,
        ModificadoPor = @ModificadoPor,
        FechaModificacion = @FechaModificacion
    WHERE Id = @Id AND Estatus = 1
    
    IF @@ROWCOUNT > 0
        SELECT @Id
    ELSE
        SELECT 0
END
GO

-- 5. Asignar rol a un usuario
CREATE PROCEDURE [dbo].[AsignarRolUsuario]
(
    @UsuarioId BIGINT,
    @RolId BIGINT,
    @AsignadoPor NVARCHAR(25),
    @EmpresaId BIGINT
)
AS
BEGIN
    -- Validar que el usuario que asigna sea administrador
    IF NOT EXISTS(
        SELECT 1 
        FROM Usuarios u
        INNER JOIN UsuarioRol ur ON u.Id = ur.UsuarioId
        INNER JOIN Rol r ON ur.RolId = r.Id
        WHERE u.NombreUsuario = @AsignadoPor 
            AND u.EmpresaId = @EmpresaId
            AND u.Estatus = 1
            AND r.Nombre = 'Administrador'
            AND ur.Estatus = 1
    )
    BEGIN
        SELECT 0
        RETURN
    END

    -- Validar que el usuario destino pertenezca a la misma empresa
    IF NOT EXISTS(
        SELECT 1 
        FROM Usuarios 
        WHERE Id = @UsuarioId 
            AND EmpresaId = @EmpresaId
            AND Estatus = 1
    )
    BEGIN
        SELECT 0
        RETURN
    END

    -- Validar que el rol pertenezca a la empresa
    IF NOT EXISTS(
        SELECT 1 
        FROM UsuarioRol ur
        INNER JOIN Usuarios u ON ur.UsuarioId = u.Id
        WHERE ur.RolId = @RolId 
            AND u.EmpresaId = @EmpresaId
            AND ur.Estatus = 1
    )
    BEGIN
        SELECT 0
        RETURN
    END

    -- Validar que no tenga ya ese rol asignado
    IF EXISTS(
        SELECT 1 
        FROM UsuarioRol 
        WHERE UsuarioId = @UsuarioId 
            AND RolId = @RolId 
            AND Estatus = 1
    )
    BEGIN
        SELECT -1
        RETURN
    END

    INSERT INTO UsuarioRol
    (UsuarioId, RolId, CreadoPor, FechaCreacion, Estatus)
    VALUES
    (@UsuarioId, @RolId, @AsignadoPor, GETDATE(), 1)

    SELECT SCOPE_IDENTITY()
END
GO

CREATE PROCEDURE [dbo].[AutenticarUsuario]

BEGIN
    SELECT u.*, 
           s.Nombre as SucursalNombre,

-- 7. Eliminar rol de un usuario (borrado lógico)
CREATE PROCEDURE [dbo].[EliminarRolUsuario]
(
    @UsuarioRolId BIGINT,
    @ModificadoPor NVARCHAR(25),
    @EmpresaId BIGINT
)
AS
BEGIN
    -- Validar que el usuario que modifica sea administrador
    IF NOT EXISTS(
        SELECT 1 
        FROM Usuarios u
        INNER JOIN UsuarioRol ur ON u.Id = ur.UsuarioId
        INNER JOIN Rol r ON ur.RolId = r.Id
        WHERE u.NombreUsuario = @ModificadoPor 
            AND u.EmpresaId = @EmpresaId
            AND u.Estatus = 1
            AND r.Nombre = 'Administrador'
            AND ur.Estatus = 1
    )
    BEGIN
        SELECT 0
        RETURN
    END

    -- No permitir eliminar el rol de Administrador del usuario administrador
    IF EXISTS(
        SELECT 1 
        FROM UsuarioRol ur
        INNER JOIN Usuarios u ON ur.UsuarioId = u.Id
        INNER JOIN Rol r ON ur.RolId = r.Id
        WHERE ur.Id = @UsuarioRolId 
            AND u.EmpresaId = @EmpresaId
            AND r.Nombre = 'Administrador'
            AND ur.Estatus = 1
    )
    BEGIN
        SELECT -1
        RETURN
    END

    UPDATE UsuarioRol
    SET Estatus = 0,
        ModificadoPor = @ModificadoPor,
        FechaModificacion = GETDATE()
    WHERE Id = @UsuarioRolId

    SELECT @UsuarioRolId
END
GO

-- 3. Guardar o actualizar responsable de categoría
CREATE PROCEDURE [dbo].[GuardarOActualizarCategoriaResponsable]
(
    @Id BIGINT,
    @CategoriaId BIGINT,
    @UsuarioId BIGINT,
    @EsPrincipal BIT,
    @CreadoPor NVARCHAR(25),
    @FechaCreacion DATETIME,
    @ModificadoPor NVARCHAR(25) = NULL,
    @FechaModificacion DATETIME = NULL,
    @Estatus BIT,
    @Usuario NVARCHAR(25)
)
AS
BEGIN
    DECLARE @EmpresaId BIGINT
    SELECT @EmpresaId = EmpresaId FROM Usuarios WHERE NombreUsuario = @Usuario AND Estatus = 1

    -- Validar que el usuario responsable pertenezca a la empresa
    IF NOT EXISTS(
        SELECT 1 
        FROM Usuarios 
        WHERE Id = @UsuarioId 
            AND EmpresaId = @EmpresaId 
            AND Estatus = 1
    )
    BEGIN
        SELECT 0
        RETURN
    END

    -- Validar que el usuario responsable pueda atender tickets
    IF NOT EXISTS(
        SELECT 1 
        FROM UsuarioRol ur
        INNER JOIN Rol r ON ur.RolId = r.Id
        WHERE ur.UsuarioId = @UsuarioId
            AND r.PuedeAtenderTickets = 1
            AND r.Estatus = 1
            AND ur.Estatus = 1
    )
    BEGIN
        SELECT -1
        RETURN
    END

    -- Validar que la categoría pertenezca a la empresa
    IF NOT EXISTS(
        SELECT 1 
        FROM Categoria c
        INNER JOIN Area a ON c.AreaId = a.Id
        INNER JOIN Usuarios u ON a.CreadoPor = u.NombreUsuario
        WHERE c.Id = @CategoriaId
            AND u.EmpresaId = @EmpresaId
            AND c.Estatus = 1
    )
    BEGIN
        SELECT 0
        RETURN
    END

    IF EXISTS(SELECT * FROM CategoriaResponsable WHERE Id = @Id)
    BEGIN
        UPDATE CategoriaResponsable
        SET CategoriaId = @CategoriaId,
            UsuarioId = @UsuarioId,
            EsPrincipal = @EsPrincipal,
            ModificadoPor = @ModificadoPor,
            FechaModificacion = @FechaModificacion,
            Estatus = @Estatus
        WHERE Id = @Id

        SELECT @Id
    END
    ELSE
    BEGIN
        -- Si se marca como principal, desmarcar otros principales de la misma categoría
        IF @EsPrincipal = 1
        BEGIN
            UPDATE CategoriaResponsable
            SET EsPrincipal = 0
            WHERE CategoriaId = @CategoriaId AND Estatus = 1
        END

        INSERT INTO CategoriaResponsable
        (CategoriaId, UsuarioId, EsPrincipal, CreadoPor, FechaCreacion, Estatus)
        VALUES
        (@CategoriaId, @UsuarioId, @EsPrincipal, @CreadoPor, @FechaCreacion, @Estatus)

        SELECT SCOPE_IDENTITY()
    END
END

            PuedeEditar = @PuedeEditar,
            PuedeEliminar = @PuedeEliminar,
            PuedeExportar = @PuedeExportar,
            ModificadoPor = @ModificadoPor,
            FechaModificacion = GETDATE()
        WHERE RolId = @RolId AND PaginaId = @PaginaId AND Estatus = 1

        SELECT 1
    END
    ELSE
    BEGIN
        -- Insertar nuevo
        INSERT INTO RolPaginaAccion
        (RolId, PaginaId, PuedeLeer, PuedeCrear, PuedeEditar, PuedeEliminar, PuedeExportar, CreadoPor, FechaCreacion, Estatus)

        (@RolId, @PaginaId, @PuedeLeer, @PuedeCrear, @PuedeEditar, @PuedeEliminar, @PuedeExportar, @ModificadoPor, GETDATE(), 1)

CREATE PROCEDURE [dbo].[ObtenerPaginasPorUsuario]
(
    @Usuario NVARCHAR(25)
)
AS
BEGIN
    -- Obtener páginas hijas con acceso (con dirección)
    WITH PaginasConAcceso AS
    (
        SELECT DISTINCT p.*
        FROM Pagina p
        INNER JOIN RolPaginaAccion rpa ON p.Id = rpa.PaginaId
        INNER JOIN Rol r ON rpa.RolId = r.Id
        INNER JOIN UsuarioRol ur ON r.Id = ur.RolId
        INNER JOIN Usuarios u ON ur.UsuarioId = u.Id
        WHERE u.NombreUsuario = @Usuario
            AND u.Estatus = 1
            AND ur.Estatus = 1
            AND r.Estatus = 1
            AND rpa.Estatus = 1
            AND p.Estatus = 1
            AND rpa.PuedeLeer = 1
            AND p.Direccion IS NOT NULL
            AND p.Direccion != ''
    )
    SELECT DISTINCT p.*
    FROM Pagina p
    WHERE p.Id IN (SELECT Id FROM PaginasConAcceso)
        OR p.Id IN (SELECT PermisosPadreId FROM PaginasConAcceso WHERE PermisosPadreId IS NOT NULL)
    ORDER BY p.OrdenB, p.Nombre
END
GO

        AND p.Estatus = 1
    ORDER BY p.Nombre

CREATE PROCEDURE [dbo].[ObtenerPermisosPorUsuario]
(
    @NombreUsuario NVARCHAR(25)
)
AS
BEGIN
    -- Obtener permisos del usuario directamente por nombre de usuario
    SELECT 
        p.Id AS PaginaId,
        p.Nombre AS PaginaNombre,
        p.Direccion,
        MAX(CASE WHEN rpa.PuedeLeer = 1 THEN 1 ELSE 0 END) AS PuedeLeer,
        MAX(CASE WHEN rpa.PuedeCrear = 1 THEN 1 ELSE 0 END) AS PuedeCrear,
        MAX(CASE WHEN rpa.PuedeEditar = 1 THEN 1 ELSE 0 END) AS PuedeEditar,
        MAX(CASE WHEN rpa.PuedeEliminar = 1 THEN 1 ELSE 0 END) AS PuedeEliminar,
        MAX(CASE WHEN rpa.PuedeExportar = 1 THEN 1 ELSE 0 END) AS PuedeExportar
    FROM Usuarios u
    INNER JOIN UsuarioRol ur ON u.Id = ur.UsuarioId
    INNER JOIN Rol r ON ur.RolId = r.Id
    INNER JOIN RolPaginaAccion rpa ON r.Id = rpa.RolId
    INNER JOIN Pagina p ON rpa.PaginaId = p.Id
    WHERE u.NombreUsuario = @NombreUsuario
        AND u.Estatus = 1
        AND ur.Estatus = 1
        AND r.Estatus = 1
        AND rpa.Estatus = 1
        AND p.Estatus = 1
        AND p.Direccion IS NOT NULL  -- Solo páginas con URL
    GROUP BY p.Id, p.Nombre, p.Direccion
    ORDER BY p.Nombre
END

-- 4. Verificar si un usuario tiene acceso a una página específica
CREATE PROCEDURE [dbo].[ValidarAccesoPagina]
(
    @Usuario NVARCHAR(25),
    @Direccion NVARCHAR(250)
)
AS
BEGIN
    SELECT CASE 
        WHEN EXISTS(
            SELECT 1 
            FROM UsuarioPagina up
            INNER JOIN Usuarios u ON up.UsuarioID = u.Id
            INNER JOIN Pagina p ON up.PaginaId = p.Id
            WHERE u.NombreUsuario = @Usuario
                AND u.Estatus = 1
                AND up.Estatus = 1
                AND p.Estatus = 1
                AND p.Direccion = @Direccion
        ) THEN 1
        ELSE 0
    END AS TieneAcceso
END
GO

-- 5. Validar permiso específico para un usuario, página y acción
CREATE PROCEDURE [dbo].[ValidarPermisoUsuario]
(
    @UsuarioId BIGINT,
    @PaginaId BIGINT,
    @Accion NVARCHAR(20)  -- 'Leer', 'Crear', 'Editar', 'Eliminar', 'Exportar'
)
AS
BEGIN
    DECLARE @TienePermiso BIT = 0

    SELECT @TienePermiso = 
        CASE 
            WHEN @Accion = 'Leer' THEN MAX(CASE WHEN rpa.PuedeLeer = 1 THEN 1 ELSE 0 END)
            WHEN @Accion = 'Crear' THEN MAX(CASE WHEN rpa.PuedeCrear = 1 THEN 1 ELSE 0 END)
            WHEN @Accion = 'Editar' THEN MAX(CASE WHEN rpa.PuedeEditar = 1 THEN 1 ELSE 0 END)
            WHEN @Accion = 'Eliminar' THEN MAX(CASE WHEN rpa.PuedeEliminar = 1 THEN 1 ELSE 0 END)
            WHEN @Accion = 'Exportar' THEN MAX(CASE WHEN rpa.PuedeExportar = 1 THEN 1 ELSE 0 END)
            ELSE 0
        END
    FROM Usuarios u
    INNER JOIN UsuarioRol ur ON u.Id = ur.UsuarioId
    INNER JOIN Rol r ON ur.RolId = r.Id
    INNER JOIN RolPaginaAccion rpa ON r.Id = rpa.RolId
    WHERE u.Id = @UsuarioId
        AND rpa.PaginaId = @PaginaId
        AND u.Estatus = 1
        AND ur.Estatus = 1
        AND r.Estatus = 1
        AND rpa.Estatus = 1

    SELECT ISNULL(@TienePermiso, 0) AS TienePermiso
END
